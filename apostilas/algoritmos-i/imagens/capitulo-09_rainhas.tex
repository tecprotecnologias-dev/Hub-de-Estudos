% \tabuleiro{x}{rainhas: linha/coluna,...}{linha sem casa livre (-1 se nenhuma)}{legenda}
\newcommand{\tabuleiro}[4]{%
  \foreach \l in {0,...,3} {
    \foreach \c in {0,...,3} {
      \pgfmathtruncatemacro{\cor}{mod(\l + \c, 2)}
      \ifnum\cor=0
        \fill[azul!50] (#1 + \c*0.6, -\l*0.6) rectangle ++(0.6, 0.6);
      \else
        \fill[white] (#1 + \c*0.6, -\l*0.6) rectangle ++(0.6, 0.6);
      \fi
    }
  }
  \draw[navy, line width=0.8pt] (#1, -1.8) rectangle ++(2.4, 2.4);
  \foreach \l/\c in {#2} {
    \node[rainha] at (#1 + \c*0.6 + 0.3, -\l*0.6 + 0.3) {R};
  }
  \ifnum#3>-1
    \foreach \c in {0,...,3} {
      \node[bloqueada] at (#1 + \c*0.6 + 0.3, -#3*0.6 + 0.3) {$\times$};
    }
  \fi
  \node[legenda, align=center, text width=30mm] at (#1 + 1.2, -2.45) {#4};
}

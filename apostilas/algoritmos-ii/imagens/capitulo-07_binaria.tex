% \linha{y}{início}{fim}{meio}{texto}
\newcommand{\linha}[5]{%
  \foreach \x [count=\i from 0] in {3, 8, 12, 17, 21, 25, 30, 34, 39, 42, 47, 51, 56, 60, 64} {
    \ifnum\i<#2
      \node[fora] (c\i) at (\i*0.8, #1) {\x};
    \else\ifnum\i>#3
      \node[fora] (c\i) at (\i*0.8, #1) {\x};
    \else\ifnum\i=#4
      \node[meio] (c\i) at (\i*0.8, #1) {\x};
    \else
      \node[casa, fill=laranja!10] (c\i) at (\i*0.8, #1) {\x};
    \fi\fi\fi
  }
  \node[explica] at (-0.4, #1+0.4) {#5};
}
